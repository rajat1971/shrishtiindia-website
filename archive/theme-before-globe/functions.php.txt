function shrishti_handle_enquiry() {
	$back = wp_get_referer() ? wp_get_referer() : home_url( '/' );
	$back = remove_query_arg( 'enquiry', strtok( $back, '#' ) );
	if ( ! isset( $_POST['sh_nonce'] ) || ! wp_verify_nonce( $_POST['sh_nonce'], 'shrishti_enquiry' ) || ! empty( $_POST['website'] ) ) {
		wp_safe_redirect( add_query_arg( 'enquiry', 'error', $back ) . '#contact' );
		exit;
	}
	$name    = sanitize_text_field( wp_unslash( $_POST['name'] ?? '' ) );
	$company = sanitize_text_field( wp_unslash( $_POST['company'] ?? '' ) );
	$phone   = sanitize_text_field( wp_unslash( $_POST['phone'] ?? '' ) );
	$email   = sanitize_email( wp_unslash( $_POST['email'] ?? '' ) );
	$service = sanitize_text_field( wp_unslash( $_POST['service'] ?? '' ) );
	$message = sanitize_textarea_field( wp_unslash( $_POST['message'] ?? '' ) );
	if ( ! $name || ! $phone || ! is_email( $email ) ) {
		wp_safe_redirect( add_query_arg( 'enquiry', 'error', $back ) . '#contact' );
		exit;
	}
	$body  = "New enquiry from shrishtiindia.com\n\n";
	$body .= "Name: $name\nCompany: $company\nPhone: $phone\nEmail: $email\nService: $service\n\nMessage:\n$message\n";
	$sent = wp_mail( SHRISHTI_ENQUIRY_TO, "Website enquiry: $service ($name)", $body, array( 'Reply-To: ' . $name . ' <' . $email . '>' ) );
	// Keep a copy in WordPress in case email delivery fails.
	$log   = get_option( 'shrishti_enquiries', array() );
	$log[] = compact( 'name', 'company', 'phone', 'email', 'service', 'message' ) + array( 'date' => current_time( 'mysql' ), 'mailed' => $sent );
	update_option( 'shrishti_enquiries', array_slice( $log, -500 ), false );
	wp_safe_redirect( add_query_arg( 'enquiry', 'sent', $back ) . '#contact' );
	exit;
}
add_action( 'admin_post_nopriv_shrishti_enquiry', 'shrishti_handle_enquiry' );
add_action( 'admin_post_shrishti_enquiry', 'shrishti_handle_enquiry' );

		$log = array_reverse( get_option( 'shrishti_enquiries', array() ) );
		echo '<div class="wrap"><h1>Website enquiries</h1><table class="widefat striped"><thead><tr><th>Date</th><th>Name</th><th>Company</th><th>Phone</th><th>Email</th><th>Service</th><th>Message</th><th>Emailed</th></tr></thead><tbody>';
		if ( ! $log ) { echo '<tr><td colspan="8">No enquiries yet.</td></tr>'; }
		foreach ( $log as $e ) {
			printf( '<tr><td>%s</td><td>%s</td><td>%s</td><td>%s</td><td>%s</td><td>%s</td><td>%s</td><td>%s</td></tr>', esc_html( $e['date'] ), esc_html( $e['name'] ), esc_html( $e['company'] ), esc_html( $e['phone'] ), esc_html( $e['email'] ), esc_html( $e['service'] ), nl2br( esc_html( $e['message'] ) ), $e['mailed'] ? 'Yes' : 'No' );
		}
		echo '</tbody></table></div>';
	}, 'dashicons-email-alt', 3 );
} );
